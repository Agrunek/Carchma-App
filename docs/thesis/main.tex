%!LW recipe=latexmk

% =================================================================================================
% Define report document class
% =================================================================================================

\documentclass[a4paper, 12pt, twoside]{report}

% =================================================================================================
% Include packages
% =================================================================================================

\usepackage[polish]{babel} % Set polish language
\usepackage{setspace} % Allow modifications in document spacing
\usepackage{geometry} % Allow modifications in document geometry
\usepackage{fontspec} % Allow modifications of available fonts
\usepackage{titlesec} % Allow modifications of headings
\usepackage{fancyhdr} % Add page numbering customization

% =================================================================================================
% Configure document
% =================================================================================================

% Set line spacing to 1.5
\onehalfspacing

% Set required margins
\geometry{includeheadfoot, vmargin=2.5cm, inner=3cm, outer=2cm, head=1.25cm, foot=1.25cm, heightrounded}

% Set paragraph indentation to 0.5 cm
\setlength{\parindent}{0.5cm}

% Set default font to Helvetica with Arial as a fallback
\IfFontExistsTF{Helvetica}{\setmainfont{Helvetica}}{\setmainfont{Arial}}

% Set style for headings
\titleformat{\chapter}{\normalfont \bfseries \fontsize{16}{19}\selectfont}{\thechapter}{1em}{}
\titleformat{\section}{\normalfont \bfseries \fontsize{14}{17}\selectfont}{\thesection}{1em}{}
\titleformat{\subsection}{\normalfont \bfseries \fontsize{13}{16}\selectfont}{\thesubsection}{1em}{}

\titlespacing*{\chapter}{0pt}{3.25ex plus 1ex minus .2ex}{1em}
\titlespacing*{\section}{0pt}{3.25ex plus 1ex minus .2ex}{1em}
\titlespacing*{\subsection}{0pt}{3.25ex plus 1ex minus .2ex}{1em}

% Add page numbers
\pagestyle{fancy}
\fancyhf{}

\fancypagestyle{plain}{
    \fancyhf{}
    \fancyfoot[LE,RO]{\thepage}
    \renewcommand{\headrulewidth}{0pt}
    \renewcommand{\footrulewidth}{0pt}
}

% =================================================================================================
% Document content
% =================================================================================================

\title{Example Title}
\author{Example Author}
\date{\today}

\begin{document}

\maketitle
\tableofcontents

\end{document}
